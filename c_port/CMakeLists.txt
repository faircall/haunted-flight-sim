cmake_minimum_required(VERSION 3.20)
project(haunted_flight_native C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(RAYLIB_ROOT "C:/raylib/raylib" CACHE PATH "raylib 5.0 source/install root")
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -ffp-contract=off)
  # Some local toolchain presets leave CMAKE_C_FLAGS_RELEASE empty.
  add_compile_options($<$<CONFIG:Release>:-O2> $<$<CONFIG:Release>:-DNDEBUG>)
endif()
add_library(hf_visibility SHARED src/visibility.c)
set_target_properties(hf_visibility PROPERTIES PREFIX "")
add_library(hf_checksum SHARED src/checksum.c)
set_target_properties(hf_checksum PROPERTIES PREFIX "")
add_library(hf_scene SHARED src/arena.c src/native_scene.c src/visibility.c)
set_target_properties(hf_scene PROPERTIES PREFIX "" WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_executable(hf_arena_tests src/arena.c src/arena_tests.c)
enable_testing()
add_test(NAME arena_lifetimes COMMAND hf_arena_tests)
add_executable(hf_allocation_audit_tests src/allocation_audit.c src/arena.c src/allocation_audit_tests.c)
target_compile_options(hf_allocation_audit_tests PRIVATE -fno-builtin-malloc -fno-builtin-calloc -fno-builtin-realloc -fno-builtin-free)
target_link_options(hf_allocation_audit_tests PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc -Wl,--wrap=free)
add_test(NAME frame_allocation_audit COMMAND hf_allocation_audit_tests)
if(NOT WIN32)
  target_link_libraries(hf_visibility PRIVATE m)
  target_link_libraries(hf_scene PRIVATE m)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/main.c")
  add_executable(haunted_native src/main.c src/visibility.c src/arena.c src/native_scene.c src/allocation_audit.c)
  target_link_options(haunted_native PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc -Wl,--wrap=free)
  target_include_directories(haunted_native PRIVATE "${RAYLIB_ROOT}/src")
  target_link_libraries(haunted_native PRIVATE "${RAYLIB_ROOT}/src/libraylib.a")
  if(WIN32)
    target_link_libraries(haunted_native PRIVATE opengl32 gdi32 winmm)
  else()
    target_link_libraries(haunted_native PRIVATE m pthread dl)
  endif()
endif()
